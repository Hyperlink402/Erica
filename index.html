// index.html 내부 기사 클릭 처리 함수 수정 부분 headerEl.addEventListener('click', async () => { card.classList.toggle('active'); if (card.classList.contains('active') && !detailLoaded && !isFetchingDetail) { isFetchingDetail = true; try { // link 파라미터도 함께 전달하여 정확도 및 속도 개선 const response = await fetch(`/api/generate-detail?id=${item.id}&title=${encodeURIComponent(item.title)}&link=${encodeURIComponent(item.link || '')}`); if (!response.ok) { const errData = await response.json().catch(() => ({})); throw new Error(errData.error || '상세 정보 로드 실패'); } const detail = await response.json(); // undefined 대비 기본값 설정 const detailsText = detail.details || detail.summary || '상세 내용이 없습니다.'; const sourcesList = detail.sources || (detail.source ? [{ name: '원문 링크', url: detail.source }] : []); const sourcesHtml = sourcesList.length > 0 ? sourcesList.map(src => `🔗 ${src.name || '출처'}`).join('') : '출처 정보 없음'; detailWrapper.innerHTML = `

상세 브리핑

${detailsText}

기사 출처

${sourcesHtml}
`; detailLoaded = true; } catch (error) { console.error("Detail Fetch Error:", error); detailWrapper.innerHTML = `

${error.message || '상세 정보를 불러올 수 없습니다.'}

`; } finally { isFetchingDetail = false; } } });
